for step in range(1, MAX_STEPS + 1):
    decision = brain(goal, history)          # 생각: 다음 행동 결정
    if "answer" in decision:
        return decision["answer"]            # 최종 답변
    result = TOOLS[decision["tool"]](**decision["args"])   # 행동: 도구 호출
